Source code for fips.problem

"""
Inverse problems: the inputs of a solve, on their labels.

A :class:`Problem` holds the prior, the observations, the forward map, the
two covariances and an optional offset of an inverse problem,
``obs = forward(state) + offset + error``. The forward map is an
:class:`~fips.Operator` (a linear problem, ``forward @ state``) or a
:class:`~fips.ForwardModel` the caller writes, with its Jacobian. It checks
their labels once, when it is built, and puts every input on the prior's and
the observations' spaces: :func:`fips.solve` then computes by position.

The problem is the question, and the :class:`~fips.Posterior` that
:func:`fips.solve` returns is the answer. The problem holds what needs no
solve and no ``m x m`` matrix (:meth:`Problem.predict`, and
:meth:`Problem.simulate` for a synthetic truth and its observations,
:meth:`Problem.ensemble` for members drawn from the prior), and makes other
problems from itself (:meth:`Problem.aggregate`, :meth:`Problem.restrict`);
the posterior holds what depends on the solve or on the ``m x m`` matrix it
factors.
"""

import logging
import numbers
from functools import cached_property
from typing import Any

import numpy as np
import pandas as pd

from fips.covariance import Covariance, CovarianceModel
from fips.forward import ForwardModel, _Aggregated, _differentiated, _evaluated
from fips.linalg import LinearOperator
from fips.operator import Operator
from fips.space import Space
from fips.vector import Vector

logger = logging.getLogger(__name__)


def _vector(value: Any, what: str) -> Vector:
    """Return `value` if it is a Vector; raise a TypeError naming the way in otherwise."""
    if not isinstance(value, Vector):
        raise TypeError(
            f"{what} must be a Vector, got {type(value).__name__}. Build one with "
            "Vector.from_series(series, block=...) or Vector.from_blocks({...})."
        )
    return value


def _covariance(value: Any, space: Space, what: str, of: str) -> Covariance:
    """
    Return `value` as a covariance on `space`.

    A :class:`~fips.covariance.CovarianceModel` is built there; a
    :class:`~fips.Covariance` must have exactly the space's labels.
    """
    if isinstance(value, CovarianceModel):
        return value.build(space, name=what)
    if not isinstance(value, Covariance):
        raise TypeError(
            f"{what} must be a Covariance or a covariance model, got "
            f"{type(value).__name__}. Make one with a model "
            "(Independent().with_std(std)), Covariance.diagonal(variances, space) "
            "or Covariance.from_frame(frame)."
        )
    return Covariance._computed(
        value._on(space, space, what, (of, of)), space, name=value.name
    )

class Problem:
    r"""
    An inverse problem: what is known before the observations, and the observations.

    The state is :math:`x \sim N(x_0, S_0)` (the prior), and the observations
    are :math:`z = H x + c + e` with :math:`e \sim N(0, S_z)`: a linear
    problem, whose forward operator :math:`H` is an
    :class:`~fips.Operator`. With a forward model, a
    :class:`~fips.ForwardModel` :math:`F` the caller writes with its Jacobian,
    they are :math:`z = F(x) + c + e`. Every input is labeled, and the labels
    are checked once, here, by an exact join: the forward operator or the
    model, the covariances and the offset must have exactly the labels of the
    prior or of the observations, in any order. Each is then put in the order
    of the prior's and the observations' spaces (a model is kept as it is,
    and its inputs and outputs are put in its order when it is called). A
    problem is not changed after it is built; :func:`fips.solve` solves it.

    The problem is the question: it holds what needs no solve and no
    ``m x m`` matrix, such as :meth:`predict`, and it makes other questions
    from itself (:meth:`aggregate`, :meth:`restrict`). What depends on the
    solve, or on the ``m x m`` innovation covariance, is on the
    :class:`~fips.Posterior`.

    Parameters
    ----------
    prior : Vector
        The prior state :math:`x_0`. Its space labels the state.
    prior_cov : Covariance or CovarianceModel
        The prior error covariance :math:`S_0`. A covariance model (such as
        ``Kronecker(...).with_std(std)``) is built on the prior's space.
    obs : Vector
        The observations :math:`z`. Its space labels the observations.
    obs_cov : Covariance or CovarianceModel
        The model-data mismatch :math:`S_z`, the covariance of the
        observation error; a model is built on the observations' space. It
        must be positive-definite.
    forward : Operator or ForwardModel
        The forward operator :math:`H`, from the state (its columns) to the
        observations (its rows); or a forward model :math:`F` with its
        Jacobian (:class:`~fips.ForwardModel`), whose rows have the
        observations' labels and whose columns the prior's. A problem with a
        model is solved by ``fips.solve(problem, "gauss_newton")``.
    offset : Vector or float, optional
        The part of the observations the state does not explain, :math:`c`
        (a background): a vector on the observations' labels, or one value
        for all of them. None (the default) is no offset.

    Raises
    ------
    TypeError
        If an input is not of a type named above.
    ValueError
        If the forward operator or model, a covariance or the offset does not
        have exactly the labels of the space it is put on: a block or a label
        on one side and not the other. The message names them. Also if every
        variance of the prior error covariance is zero (some may be: an
        element known exactly, a bias block given as 0).

    See Also
    --------
    fips.solve : Solve a problem; it returns a :class:`~fips.Posterior`.
    fips.ForwardModel : A forward map that is not linear, with its Jacobian.

    Notes
    -----
    A forward operator that covers only part of the state (a transport
    Jacobian has columns only for the cells a footprint touched) is placed on
    the whole state by :meth:`Operator.from_blocks <fips.Operator.from_blocks>`,
    which puts a labeled block by its labels: a cell the Jacobian lacks is a
    column of zeros, and a label the space lacks raises. With ``J`` the
    labeled Jacobian (a DataFrame, dense or pandas-sparse, or a tuple
    ``(matrix, row_labels, col_labels)``)::

        forward = Operator.from_blocks(
            {("co2", "flux"): J}, rows=obs.space, cols=prior.space
        )

    Examples
    --------
    >>> import numpy as np
    >>> import pandas as pd
    >>> import fips
    >>> cells = pd.Index(["a", "b"], name="cell")
    >>> obs_id = pd.Index(["o1", "o2", "o3"], name="obs_id")
    >>> prior = fips.Vector.from_series(pd.Series(0.0, index=cells), block="flux")
    >>> obs = fips.Vector.from_series(
    ...     pd.Series([1.0, 2.0, 3.0], index=obs_id), block="co2"
    ... )
    >>> problem = fips.Problem(
    ...     prior=prior,
    ...     prior_cov=fips.Independent().with_std(1.0),
    ...     obs=obs,
    ...     obs_cov=fips.Independent().with_std(np.sqrt(0.1)),
    ...     forward=fips.Operator(
    ...         np.array([[1.0, 0.0], [0.0, 1.0], [1.0, 1.0]]), obs.space, prior.space
    ...     ),
    ... )
    >>> problem
    Problem(state: Space(flux: 2 x ['cell']), obs: Space(co2: 3 x ['obs_id']))
    >>> fips.solve(problem).estimate.values.round(3)
    array([0.997, 1.906])
    """

    def __init__(
        self,
        prior: Vector,
        prior_cov: "Covariance | CovarianceModel",
        obs: Vector,
        obs_cov: "Covariance | CovarianceModel",
        forward: "Operator | ForwardModel",
        offset: "Vector | float | None" = None,
    ):
        self._prior = _vector(prior, "prior")
        self._obs = _vector(obs, "obs")
        state, observed = self._prior.space, self._obs.space

        self._forward: Operator | ForwardModel
        if isinstance(forward, Operator):
            self._forward = Operator(
                forward._on(
                    observed, state, "forward", ("the observations", "the prior")
                ),
                observed,
                state,
                name=forward.name,
            )
        elif isinstance(forward, ForwardModel):
            try:
                rows, cols = forward.rows, forward.cols
            except AttributeError:
                raise TypeError(
                    f"The forward model {type(forward).__name__} has no rows or "
                    "cols: its __init__ must call super().__init__(rows, cols)."
                ) from None
            observed._align(
                rows, "the rows of forward", "The rows of forward and the observations"
            )
            state._align(
                cols, "the columns of forward", "The columns of forward and the prior"
            )
            self._forward = forward
        else:
            raise TypeError(
                f"forward must be an Operator or a ForwardModel, got "
                f"{type(forward).__name__}. Build an operator with Operator(matrix, "
                "rows, cols), Operator.from_blocks or Operator.from_frame; a map "
                "that is not linear is a subclass of fips.ForwardModel."
            )
        self._prior_cov = _covariance(prior_cov, state, "prior_cov", "the prior")
        if len(state) and not self._prior_cov.data.diagonal().any():
            raise ValueError(
                "Every variance of prior_cov is zero, which says every state "
                "element is known exactly: no observation could change the prior. "
                "Check its standard deviations and their units. Some zero "
                "variances are fine; to fix the whole state, use Problem.predict."
            )
        self._obs_cov = _covariance(obs_cov, observed, "obs_cov", "the observations")

        if offset is None:
            self._offset = None
        elif isinstance(offset, Vector):
            positions = observed._align(
                offset.space, "the offset", "The offset and the observations"
            )
            self._offset = Vector(offset.values[positions], observed, name=offset.name)
        elif isinstance(offset, (int, float, np.integer, np.floating)):
            self._offset = Vector(np.full(len(observed), float(offset)), observed)
        else:
            raise TypeError(
                f"offset must be a Vector, a number or None, got {type(offset).__name__}."
            )

    @property
    def prior(self) -> Vector:
        """The prior state, :math:`x_0`; its space labels the state."""
        return self._prior

    @property
    def prior_cov(self) -> Covariance:
        """The prior error covariance, :math:`S_0`, on the prior's space."""
        return self._prior_cov

    @property
    def obs(self) -> Vector:
        """The observations, :math:`z`; their space labels the observations."""
        return self._obs

    @property
    def obs_cov(self) -> Covariance:
        """The model-data mismatch, :math:`S_z`, on the observations' space."""
        return self._obs_cov

    @property
    def forward(self) -> "Operator | ForwardModel":
        """The forward operator, :math:`H` (rows on the observations, columns on the prior), or the forward model, :math:`F`, as given."""
        return self._forward

    @property
    def _operator(self) -> Operator:
        """
        The forward operator, for what needs a linear forward.

        Raises
        ------
        TypeError
            If the forward is a model.
        """
        forward = self._forward
        if isinstance(forward, Operator):
            return forward
        raise TypeError(
            "This needs a linear forward operator, and the problem's forward is a "
            f"model, {type(forward).__name__}. Solve the problem with "
            'fips.solve(problem, "gauss_newton"), or linearize it at a state x '
            "with problem.linearized(x)."
        )

    @property
    def offset(self) -> Vector | None:
        """The offset, :math:`c`, on the observations' space, or None."""
        return self._offset

    def predict(self, x: Vector) -> Vector:
        r"""
        Return the observations the state `x` models, :math:`H x + c` (:math:`F(x) + c` with a forward model).

        Parameters
        ----------
        x : Vector
            A state on the prior's labels, in any order: the prior, a
            posterior estimate, a truth to test with.

        Returns
        -------
        Vector
            :math:`H x + c` (with the offset :math:`c`, if any), or
            :math:`F(x) + c` with a forward model, on the observations' space.

        Raises
        ------
        TypeError
            If `x` is not a Vector.
        ValueError
            If `x` does not have exactly the prior's labels; the message names
            those on one side only.

        Examples
        --------
        >>> import numpy as np
        >>> import pandas as pd
        >>> import fips
        >>> cells = pd.Index(["a", "b"], name="cell")
        >>> obs_id = pd.Index(["o1", "o2", "o3"], name="obs_id")
        >>> prior = fips.Vector.from_series(pd.Series(1.0, index=cells), block="flux")
        >>> obs = fips.Vector.from_series(
        ...     pd.Series([1.0, 2.0, 3.0], index=obs_id), block="co2"
        ... )
        >>> problem = fips.Problem(
        ...     prior=prior,
        ...     prior_cov=fips.Independent().with_std(1.0),
        ...     obs=obs,
        ...     obs_cov=fips.Independent().with_std(0.3),
        ...     forward=fips.Operator(
        ...         np.array([[1.0, 0.0], [0.0, 1.0], [1.0, 1.0]]),
        ...         obs.space,
        ...         prior.space,
        ...     ),
        ...     offset=0.5,
        ... )
        >>> problem.predict(problem.prior).values
        array([1.5, 1.5, 2.5])
        """
        predicted = self._apply(self._state_values(x, "x")) + self._offset_values()
        return Vector(predicted, self._obs.space)

    def linearized(self, x: Vector) -> "Problem":
        r"""
        Return the linear problem at the state `x`: the forward model replaced by its tangent there.

        With a forward model :math:`F` and its Jacobian :math:`J = J(x)`,
        :math:`F(x') \approx F(x) + J (x' - x)`, so the problem linearized at
        `x` has the forward operator :math:`J` and the offset :math:`c + F(x)
        - J x`. It predicts what the model does at `x` itself, and near `x` to
        first order. Its analytic solve, ``fips.solve(problem.linearized(x))``,
        is one step of Gauss-Newton from `x`; ``fips.solve(problem,
        "gauss_newton")`` repeats the step from the prior until the estimate
        stops moving.

        Parameters
        ----------
        x : Vector
            The state to linearize at, on the prior's labels (in any order).

        Returns
        -------
        Problem
            The linear problem (of the same class), with the same prior,
            observations and covariances. A problem whose forward is an
            operator is linear already, and is returned itself.

        Raises
        ------
        TypeError
            If `x` is not a Vector, or the model's Jacobian is not an
            Operator.
        ValueError
            If `x` does not have exactly the prior's labels, or what the model
            returns does not have exactly its labels.

        See Also
        --------
        fips.ForwardModel : The model and its Jacobian.
        fips.taylortest : Check the Jacobian against the model.

        Examples
        --------
        >>> import numpy as np
        >>> import pandas as pd
        >>> import fips
        >>> from fips.linalg import DiagonalOperator
        >>> class Square(fips.ForwardModel):
        ...     def __call__(self, x):
        ...         return fips.Vector(x.values**2, self.rows)
        ...
        ...     def jacobian(self, x):
        ...         form = DiagonalOperator(2 * x.values)
        ...         return fips.Operator(form, self.rows, x.space)
        >>> state = fips.Space({"flux": pd.Index(["a", "b"], name="cell")})
        >>> observed = fips.Space({"co2": pd.Index(["o1", "o2"], name="obs_id")})
        >>> problem = fips.Problem(
        ...     prior=fips.Vector([1.0, 2.0], state),
        ...     prior_cov=fips.Independent().with_std(1.0),
        ...     obs=fips.Vector([1.5, 3.0], observed),
        ...     obs_cov=fips.Independent().with_std(0.1),
        ...     forward=Square(observed, state),
        ...     offset=10.0,
        ... )
        >>> linear = problem.linearized(problem.prior)
        >>> linear.forward.dense()
        array([[2., 0.],
               [0., 4.]])
        >>> linear.offset.values  # c + F(x) - J x
        array([9., 6.])
        >>> linear.predict(problem.prior).values  # F(x) + c at x itself
        array([11., 14.])
        """
        if isinstance(self._forward, Operator):
            return self
        return self._linearized(self._state_values(x, "x"))



    def _linearized(
        self, values: np.ndarray, modeled: np.ndarray | None = None
    ) -> "Problem":
        r"""
        Return the problem linearized at the state `values` (in the prior's order); `modeled` is :math:`F(x)` when it is known.

        Raises
        ------
        TypeError
            If the forward is an operator.
        """
        model = self._forward
        if not isinstance(model, ForwardModel):
            raise TypeError("Only a problem with a forward model is linearized.")
        state, observed = self._prior.space, self._obs.space
        if modeled is None:
            modeled = self._apply(values)
        form, name = self._jacobian(values)
        offset = self._offset_values() + modeled - np.asarray(form @ values)
        return type(self)(
            prior=self._prior,
            prior_cov=self._prior_cov,
            obs=self._obs,
            obs_cov=self._obs_cov,
            forward=Operator(form, observed, state, name=name),
            offset=Vector(
                offset,
                observed,
                name=None if self._offset is None else self._offset.name,
            ),
        )

    def simulate(
        self,
        rng: "int | np.random.Generator | None" = None,
        truth: "Vector | None" = None,
    ) -> tuple[Vector, Vector]:
        r"""
        Return a truth and the observations it would give, :math:`H x + c + e`.

        The problem as a world that makes data: the truth is drawn from the
        prior, :math:`x \sim N(x_0, S_0)`, and the observations are its
        noise-free prediction (:meth:`predict`) plus an error drawn from the
        model-data mismatch, :math:`e \sim N(0, S_z)`. Solving the problem on
        these observations, ``post.with_data(obs)``, and comparing the estimate
        with the truth tests an inversion on data whose answer is known; a
        loop of them over one generator is a Monte Carlo of how often the
        posterior's intervals hold the truth.

        Parameters
        ----------
        rng : int, numpy.random.Generator or None, default None
            A seed, or the generator to draw from (used as given, so one
            generator passed to every call gives independent trials). None
            draws from fresh entropy. It goes through
            :func:`numpy.random.default_rng`, as SciPy's ``rng`` argument
            does. The truth is drawn first and the error second, so a given
            `truth` draws the error alone.
        truth : Vector, optional
            A state to use as the truth instead of drawing one, on the prior's
            labels in any order: another inventory, 1.5 times the prior. Its
            observations still carry the error.

        Returns
        -------
        truth : Vector
            The truth, on the prior's space (the given one, put in the prior's
            order).
        obs : Vector
            The synthetic observations, on the observations' space and named
            as the problem's. They are ready for
            :meth:`Posterior.with_data <fips.Posterior.with_data>`.

        Raises
        ------
        TypeError
            If `truth` is not a Vector.
        ValueError
            If `truth` does not have exactly the prior's labels; the message
            names those on one side only.
        numpy.linalg.LinAlgError
            If a covariance has a clearly negative eigenvalue.

        See Also
        --------
        predict : The observations without the error.
        Covariance.sample : The draw of the truth and of the error.
        Posterior.with_data : The posterior of other observations, one product.
        Posterior.recovery : The exact mean and spread of the error over such trials.

        Notes
        -----
        The problem's own observations are not used, only their labels and
        name. A different world (another background, another forward operator,
        an observation error four times the assumed one) is another
        :class:`Problem`, made the way the first was with one input changed:
        ``world.simulate(rng)`` gives its data, and ``post.with_data(obs)``
        solves them with the assumptions of `post`'s problem.

        Examples
        --------
        >>> import numpy as np
        >>> import pandas as pd
        >>> import fips
        >>> cells = pd.Index(["a", "b"], name="cell")
        >>> obs_id = pd.Index(["o1", "o2", "o3"], name="obs_id")
        >>> prior = fips.Vector.from_series(pd.Series(0.0, index=cells), block="flux")
        >>> obs = fips.Vector.from_series(pd.Series(0.0, index=obs_id), block="co2")
        >>> problem = fips.Problem(
        ...     prior=prior,
        ...     prior_cov=fips.Independent().with_std(1.0),
        ...     obs=obs,
        ...     obs_cov=fips.Independent().with_std(0.1),
        ...     forward=fips.Operator(
        ...         np.array([[1.0, 0.0], [0.0, 1.0], [1.0, 1.0]]),
        ...         obs.space,
        ...         prior.space,
        ...     ),
        ... )
        >>> truth, synthetic = problem.simulate(rng=0)
        >>> truth.values.round(2)
        array([ 0.13, -0.13])
        >>> synthetic.values.round(2)
        array([ 0.19, -0.12, -0.06])
        >>> fips.solve(problem).with_data(synthetic).estimate.values.round(2)
        array([ 0.15, -0.16])
        """
        rng = np.random.default_rng(rng)
        state = self._prior.space
        if truth is None:
            draw = self._prior_cov.sample(rng)
            truth = Vector(self._prior.values + draw.values, state)
        else:
            values = self._state_values(truth, "truth", "The truth and the prior")
            truth = Vector(values, state, name=truth.name)
        noise = self._obs_cov.sample(rng)
        obs = Vector(
            self._apply(truth.values) + self._offset_values() + noise.values,
            self._obs.space,
            name=self._obs.name,
        )
        return truth, obs

    def restrict(self, keep: Space) -> "Problem":
        r"""
        Return the problem on part of the state, the rest held at its prior.

        Use it to solve for some of the state only, such as the cells the
        observations see well, and keep every other element at its prior
        value. Which elements to keep is your choice: for example, the cells
        whose total sensitivity is above a percentile of all cells'
        (:func:`fips.aggregate.sum` of the forward operator's absolute column
        sums, grouped by cell, gives that total).

        The dropped elements still add to the observations, at their prior
        values: :math:`H_r x_{0,r}` (the forward operator's columns for them
        times their prior) is added to the offset. The prior, the prior
        covariance and the forward operator are cut to the kept labels; the
        covariance keeps its form, so a Kronecker covariance stays as its
        factors. The dropped elements' prior uncertainty is not added to the
        model-data mismatch: they are held exactly at the prior.

        After solving, ``post.estimate.combine_first(problem.prior)`` puts
        the estimate back on the whole state, with the prior for the dropped
        elements (:meth:`Vector.combine_first`).

        Parameters
        ----------
        keep : Space
            The state of the new problem: blocks of the prior's space, each
            with some of its labels (in any order; the new problem has them
            in this order). A block of the prior not in `keep` is held at the
            prior whole.

        Returns
        -------
        Problem
            The restricted problem (of the same class). The observations and
            their covariance are unchanged; its offset is this problem's
            offset (zero if none) plus the contribution of the elements left
            out.

        Raises
        ------
        TypeError
            If the forward is a model: holding elements at their prior puts
            what they add to the observations into the offset, which is a
            constant only when the forward is linear.
        ValueError
            If `keep` has a block or a label that the prior lacks.

        See Also
        --------
        Vector.combine_first : The estimate put back on the whole state.

        Examples
        --------
        >>> import numpy as np
        >>> import pandas as pd
        >>> import fips
        >>> cells = pd.Index(["a", "b", "c"], name="cell")
        >>> obs_id = pd.Index(["o1", "o2"], name="obs_id")
        >>> prior = fips.Vector.from_series(
        ...     pd.Series([1.0, 2.0, 3.0], index=cells), block="flux"
        ... )
        >>> obs = fips.Vector.from_series(
        ...     pd.Series([4.0, 5.0], index=obs_id), block="co2"
        ... )
        >>> H = np.array([[1.0, 1.0, 1e-6], [0.0, 1.0, 1e-6]])
        >>> problem = fips.Problem(
        ...     prior=prior,
        ...     prior_cov=fips.Independent().with_std(1.0),
        ...     obs=obs,
        ...     obs_cov=fips.Independent().with_std(0.5),
        ...     forward=fips.Operator(H, obs.space, prior.space),
        ... )
        >>> small = problem.restrict(fips.Space({"flux": cells[:2]}))
        >>> small.offset.values
        array([3.e-06, 3.e-06])
        >>> post = fips.solve(small)
        >>> post.estimate.combine_first(problem.prior).values.round(3)
        array([0.034, 4.207, 3.   ])
        """
        if not isinstance(keep, Space):
            raise TypeError(f"keep must be a Space, got {type(keep).__name__}.")
        if not isinstance(self._forward, Operator):
            raise TypeError(
                "restrict needs a linear forward operator, and this problem's "
                f"forward is a model, {type(self._forward).__name__}: holding "
                "elements at their prior puts what they add to the observations "
                "into the offset, which is a constant only when the forward is "
                "linear. Restrict the problem linearized at a state, "
                "problem.linearized(x).restrict(keep), or write the model on the "
                "elements to keep."
            )
        state = self._prior.space
        kept = state._subset(keep, "The space to keep")
        left_out = np.ones(len(state), dtype=bool)
        left_out[kept] = False
        H = self._forward.data
        rows = np.arange(H.shape[0])

        offset = self._offset
        if left_out.any():
            removed = np.flatnonzero(left_out)
            contribution = H._select(rows, removed) @ self._prior.values[removed]
            name = None if offset is None else offset.name
            values = self._offset_values() + contribution
            offset = Vector(values, self._obs.space, name=name)

        prior_cov = self._prior_cov
        return type(self)(
            prior=Vector(self._prior.values[kept], keep, name=self._prior.name),
            prior_cov=Covariance._computed(
                prior_cov.data._select(kept, kept), keep, name=prior_cov.name
            ),
            obs=self._obs,
            obs_cov=self._obs_cov,
            forward=Operator(
                H._select(rows, kept), self._obs.space, keep, name=self._forward.name
            ),
            offset=offset,
        )

    def aggregate(self, W: Operator) -> "Problem":
        r"""
        Return the problem of aggregated observations.

        Each input on the observations goes through the same aggregation
        :math:`W`: the observations :math:`W z`, the forward operator
        :math:`W H`, the model-data mismatch :math:`W S_z W^T` and the offset
        :math:`W c`. Each keeps the cheapest form (a sparse :math:`W` and a
        diagonal :math:`S_z` give a sparse mismatch). A forward model
        :math:`F` is followed by the aggregation: the new problem's model is
        :math:`W F(x)`, with the Jacobian :math:`W J(x)`.

        Parameters
        ----------
        W : Operator
            The aggregation, such as :func:`fips.aggregate.mean`'s: its
            columns the observations' labels (in any order), its rows the
            aggregated observations.

        Returns
        -------
        Problem
            The problem (of the same class) on the aggregated observations,
            with the same prior and prior covariance.

        Raises
        ------
        TypeError
            If `W` is not an Operator.
        ValueError
            If the columns of `W` do not have exactly the observations'
            labels.

        Examples
        --------
        >>> import numpy as np
        >>> import pandas as pd
        >>> import fips
        >>> times = pd.date_range("2020-01-01", periods=4, freq="12h", name="obs_time")
        >>> obs = fips.Vector.from_series(
        ...     pd.Series([1.0, 2.0, 3.0, 4.0], index=times), block="co2"
        ... )
        >>> prior = fips.Vector.from_series(
        ...     pd.Series(0.0, index=pd.Index(["a"], name="cell")), block="flux"
        ... )
        >>> problem = fips.Problem(
        ...     prior=prior,
        ...     prior_cov=fips.Independent().with_std(1.0),
        ...     obs=obs,
        ...     obs_cov=fips.Independent().with_std(1.0),
        ...     forward=fips.Operator(np.ones((4, 1)), obs.space, prior.space),
        ... )
        >>> W = fips.aggregate.mean(
        ...     obs.space, by=pd.Grouper(level="obs_time", freq="1D")
        ... )
        >>> daily = problem.aggregate(W)
        >>> daily.obs.values, daily.obs_cov.variances.values
        (array([1.5, 3.5]), array([0.5, 0.5]))
        """
        if not isinstance(W, Operator):
            raise TypeError(
                f"An aggregation is an Operator, not {type(W).__name__}: build one "
                "with fips.aggregate.mean or fips.aggregate.sum."
            )
        observed = self._obs.space
        form = W._on(
            W.rows, observed, "the aggregation", ("its rows", "the observations")
        )
        rows = W.rows
        offset = self._offset
        if offset is not None:
            offset = Vector(form @ offset.values, rows, name=offset.name)
        forward = self._forward
        if isinstance(forward, Operator):
            forward = Operator(
                form @ forward.data, rows, self._prior.space, name=forward.name
            )
        else:
            forward = _Aggregated(forward, Operator(form, rows, observed, name=W.name))
        return type(self)(
            prior=self._prior,
            prior_cov=self._prior_cov,
            obs=Vector(form @ self._obs.values, rows, name=self._obs.name),
            obs_cov=self._obs_cov.aggregate(W),
            forward=forward,
            offset=offset,
        )

    def ensemble(
        self, n: int, rng: "int | np.random.Generator | None" = None
    ) -> pd.DataFrame:
        r"""
        Return `n` members drawn from the prior, one column each: the first stage of the ensemble method.

        Each member is :math:`x_0 + \delta_j - \bar{\delta}`, with
        :math:`\delta_j \sim N(0, S_0)` drawn through the prior error
        covariance's root and their mean :math:`\bar{\delta}` taken out, so
        the members' mean is the prior exactly and their sample covariance,
        :math:`\sum_j (x_j - x_0)(x_j - x_0)^T / (n - 1)`, has the mean
        :math:`S_0`. Run the forward model on each member (the second stage,
        outside fips: on a cluster, one run a member), then give both to
        ``fips.solve(problem, "ensemble", members=members,
        predicted=modeled)``.

        Parameters
        ----------
        n : int
            The number of members, at least 2.
        rng : int, numpy.random.Generator or None, default None
            A seed, or the generator to draw from, as :meth:`simulate` takes
            it. The standard normals are drawn as one ``columns x n`` array.

        Returns
        -------
        pandas.DataFrame
            One row per state element, labeled as ``problem.prior.index``, and
            one column per member, a ``RangeIndex`` named ``member``.

        Raises
        ------
        ValueError
            If `n` is not an integer of at least 2.
        numpy.linalg.LinAlgError
            If the prior error covariance has a clearly negative eigenvalue.

        See Also
        --------
        fips.methods.ensemble.EnsembleOptions : The second stage's options.
        simulate : One draw of a truth and its observations.

        Examples
        --------
        >>> import numpy as np
        >>> import pandas as pd
        >>> import fips
        >>> cells = pd.Index(["a", "b"], name="cell")
        >>> prior = fips.Vector.from_series(pd.Series(1.0, index=cells), block="flux")
        >>> obs = fips.Vector(
        ...     [1.5, 0.5, 1.8], fips.Space({"co2": pd.RangeIndex(3, name="obs_id")})
        ... )
        >>> H = fips.Operator(
        ...     np.array([[1.0, 0.0], [0.0, 1.0], [1.0, 1.0]]), obs.space, prior.space
        ... )
        >>> problem = fips.Problem(
        ...     prior=prior,
        ...     prior_cov=fips.Independent().with_std(0.5),
        ...     obs=obs,
        ...     obs_cov=fips.Independent().with_std(0.2),
        ...     forward=H,
        ... )
        >>> members = problem.ensemble(100, rng=0)
        >>> members.shape
        (2, 100)
        >>> members.mean(axis=1).round(12).tolist()  # the prior, exactly
        [1.0, 1.0]
        >>> modeled = pd.DataFrame(
        ...     H.data @ members.to_numpy(), index=obs.index, columns=members.columns
        ... )
        >>> post = fips.solve(problem, "ensemble", members=members, predicted=modeled)
        >>> post.estimate.values.round(2)  # the analytic one is [1.37, 0.51]
        array([1.36, 0.51])
        """
        if not isinstance(n, numbers.Integral) or isinstance(n, bool) or n < 2:
            raise ValueError(
                f"n is the number of members, an integer of at least 2, got {n!r}."
            )
        rng = np.random.default_rng(rng)
        root = self._prior_cov.data._root
        draws = np.asarray(root @ rng.standard_normal((root.shape[1], int(n))))
        draws -= draws.mean(axis=1, keepdims=True)
        return pd.DataFrame(
            self._prior.values[:, None] + draws,
            index=self._prior.index,
            columns=pd.RangeIndex(int(n), name="member"),
        )



    def _state_values(
        self, x: Vector, what: str, pair: str | None = None
    ) -> np.ndarray:
        """Return the values of the state `x` in the prior's order; `what` names it in the errors."""
        state = self._prior.space
        values = _vector(x, what).values
        if x.space is not state:
            values = values[
                state._align(x.space, what, pair or f"{what} and the prior")
            ]
        return values

    @cached_property
    def _obs_logdet(self) -> float:
        r"""
        :math:`\ln \lvert S_z \rvert`, from the form's own factors, computed once.

        :attr:`Posterior.log_evidence <fips.Posterior.log_evidence>` needs it
        at every gamma, and the posteriors of a sweep share their problem. Not
        pickled: a problem read back computes it again.
        """
        return self._obs_cov.logdet()

    @cached_property
    def _model_order(
        self,
    ) -> tuple[np.ndarray | None, np.ndarray | None, np.ndarray | None]:
        """
        The positions that put the forward model's labels in the problem's order, found once (None where the orders are the same).

        The state's position of each of the model's columns (for
        :meth:`_to_model`), the model's row of each observation (for
        :meth:`_apply` and :meth:`_jacobian`), and the model's column of each
        state element (for :meth:`_jacobian`). Not pickled: a problem read
        back finds them again.
        """
        model = self._forward
        assert isinstance(model, ForwardModel)  # called for a model only

        def kept(positions: np.ndarray) -> np.ndarray | None:
            """Return `positions`, or None if they keep the order."""
            same = np.array_equal(positions, np.arange(len(positions)))
            return None if same else positions

        state, observed = self._prior.space, self._obs.space
        to_model = None
        to_state = None
        if model.cols is not state:
            positions = model.cols._align(state, "the prior", "The model's columns")
            to_model = kept(positions)
            to_state = None if to_model is None else np.argsort(positions)
        rows = None
        if model.rows is not observed:
            rows = kept(
                observed._align(model.rows, "the model's rows", "The model's rows")
            )
        return to_model, rows, to_state

    def _to_model(self, values: np.ndarray) -> np.ndarray:
        """Return the state `values` (in the prior's order) in the order of the forward model's columns."""
        to_model = self._model_order[0]
        return values if to_model is None else values[to_model]

    def _apply(self, values: np.ndarray) -> np.ndarray:
        """
        Return :math:`H x`, or :math:`F(x)` with a forward model, for the state values `x` (in the prior's order), without the offset.

        Raises
        ------
        TypeError, ValueError
            If the model does not return a Vector with exactly its rows' labels.
        """
        forward = self._forward
        if isinstance(forward, Operator):
            return forward.data @ values
        modeled = _evaluated(forward, self._to_model(values))
        rows = self._model_order[1]
        return modeled if rows is None else modeled[rows]

    def _gradient(self, values: np.ndarray, v: np.ndarray) -> np.ndarray:
        """
        Return :math:`J(x)^T v` for the state values `x` (in the prior's order) and `v` on the observations, in the prior's order.

        Through the model's ``_gradient``, which a model with a cheaper
        product than its Jacobian's has.
        """
        model = self._forward
        assert isinstance(model, ForwardModel)  # called for a model only
        gradient = model._gradient(self._to_model(values), self._to_model_rows(v))
        to_state = self._model_order[2]
        return gradient if to_state is None else gradient[to_state]

    def _curvature(self, values: np.ndarray, v: np.ndarray) -> np.ndarray | None:
        r"""
        Return the diagonal of :math:`\sum_k v_k \nabla^2 F_k(x)` for the state values `x` (in the prior's order) and `v` on the observations, in the prior's order; None if the model has no diagonal one.

        Through the model's ``_curvature`` (a transform's; None for any other
        model, whose steps are then Gauss-Newton's).
        """
        model = self._forward
        assert isinstance(model, ForwardModel)  # called for a model only
        curvature = model._curvature(self._to_model(values), self._to_model_rows(v))
        to_state = self._model_order[2]
        if curvature is None or to_state is None:
            return curvature
        return curvature[to_state]

    def _to_model_rows(self, v: np.ndarray) -> np.ndarray:
        """Return `v`, on the observations, in the order of the forward model's rows."""
        rows = self._model_order[1]
        if rows is None:
            return v
        on_rows = np.empty(len(v))
        on_rows[rows] = v
        return on_rows

    def _jacobian(self, values: np.ndarray) -> tuple[LinearOperator, str | None]:
        """
        Return the Jacobian of the forward model at the state `values` (in the prior's order), its rows and columns in the problem's order, and its name.

        Raises
        ------
        TypeError, ValueError
            If the model's ``jacobian`` does not return an Operator with
            exactly the model's labels.
        """
        model = self._forward
        assert isinstance(model, ForwardModel)  # called for a model only
        J = _differentiated(model, self._to_model(values))
        form = J.data
        _, rows, cols = self._model_order
        if rows is not None or cols is not None:
            m, n = form.shape
            form = form._select(
                np.arange(m) if rows is None else rows,
                np.arange(n) if cols is None else cols,
            )
        return form, J.name

    def _offset_values(self) -> np.ndarray:
        """Return the offset's values, zeros where there is none."""
        if self._offset is None:
            return np.zeros(len(self._obs))
        return self._offset.values

    def __repr__(self) -> str:
        """Return the two spaces."""
        return f"Problem(state: {self._prior.space!r}, obs: {self._obs.space!r})"

    def __getstate__(self) -> dict[str, Any]:
        """Return the inputs, already on their spaces."""
        return {
            "prior": self._prior,
            "prior_cov": self._prior_cov,
            "obs": self._obs,
            "obs_cov": self._obs_cov,
            "forward": self._forward,
            "offset": self._offset,
        }

    def __setstate__(self, state: dict[str, Any]) -> None:
        """Restore from :meth:`__getstate__`; the labels were checked when it was built."""
        self._prior = state["prior"]
        self._prior_cov = state["prior_cov"]
        self._obs = state["obs"]
        self._obs_cov = state["obs_cov"]
        self._forward = state["forward"]
        self._offset = state["offset"]
